        function getTransactionObject(id) {
            if (id === 'bank') {
                return { 
                    id: 'bank', 
                    type: 'bank', 
                    account: accounts.mainBankReserve, 
                    name: accounts.mainBankReserve.name 
                };
            }


            const [typeCode, objId] = id.split('_');


            if (typeCode === 'p') return players.find(p => p.id === objId); 


            if (typeCode === 'cc') { 
                const card = cards.credit.find(c => c.id === objId);
                return { ...card, account: accounts.mainCredit, name: `CC: ${card.holderName}` };
            }


            if (typeCode === 'dc') { 
                const card = cards.debit.find(c => c.id === objId);
                const account = accounts.debit.find(a => a.id === card.accountId);
                return { ...card, account: account, name: `DC: ${card.holderName}` };
            }
            return null;
        }

                receiver: receiverName,
                amount: REWARD_AMOUNT_CASH,
                description: `*** CASH REWARD! *** Bonus $${REWARD_AMOUNT_CASH.toLocaleString()} for transaction #${totalTransactions}`,
            });
            return `Cash Reward ($${REWARD_AMOUNT_CASH.toLocaleString()}) Paid.`;
        }


        // NEW: Function to handle the discount reward
        function processDiscountReward(receiverName) {
            const rewardType = discountToggle;
            const rewardValue = REWARD_AMOUNT_DISCOUNT;
            
            // Toggle for the next time
            discountToggle = (rewardType === 'Property Off') ? 'Site Off' : 'Property Off';
            nextRewardDiscount += 3;


            transactions.push({
                time: new Date().toLocaleTimeString(),
                sender: `The Bank`,
                receiver: receiverName,
                amount: 0, // No direct cash transfer, just a notification
                description: `*** DISCOUNT REWARD! *** ${rewardType} ($${rewardValue.toLocaleString()} value) awarded to the receiver for transaction #${totalTransactions}`,
            });
            return `${rewardType} ($${rewardValue.toLocaleString()} Off) Awarded.`;
        }




        function executeTransaction() {
            const sourceId = document.getElementById('source').value;
            const destinationId = document.getElementById('destination').value;
            const amountInput = document.getElementById('amount').value;
            let description = document.getElementById('description').value || 'General Transfer';
            
            const amount = parseFloat(amountInput);


            if (isNaN(amount) || amount <= 0) {
                alert("Please enter a valid amount greater than zero.");
                return;
            }
            if (sourceId === destinationId) {
                alert("Source and Destination must be different.");
                return;
            }


            const sourceObj = getTransactionObject(sourceId);
            const destinationObj = getTransactionObject(destinationId);


            let sourceAccount = sourceObj.account || sourceObj;
            let destinationAccount = destinationObj.account || destinationObj;
            
            if (sourceAccount.balance < amount) {
                 alert(`${sourceObj.name} (Account: ${sourceAccount.name}) does not have enough funds. Current Balance: $${sourceAccount.balance.toLocaleString()}`);
                return;
            }


            // 1. Perform the transfer
            sourceAccount.balance -= amount;
            destinationAccount.balance += amount;


            // 2. Record the base transaction
            transactions.push({
                time: new Date().toLocaleTimeString(),
                sender: sourceObj.name,
                receiver: destinationObj.name,
                amount: amount,
                description: description,
            });


            // 3. Update total transaction count
            totalTransactions++;
            
            let alertMessage = `Transaction successful: $${amount.toLocaleString()} transferred.`;
            
            // 4. Check for reward milestones
            const isCashReward = (totalTransactions === nextRewardCash);
            const isDiscountReward = (totalTransactions === nextRewardDiscount);


            if (isCashReward) {
                alertMessage += `\n\n🎉 ${processCashReward(destinationAccount, destinationObj.name)}`;
            }


            if (isDiscountReward) {
                alertMessage += `\n\n🎁 ${processDiscountReward(destinationObj.name)}`;
            }
            
            if (isCashReward || isDiscountReward) {
                alert(alertMessage);
            }


            updateUI();
            document.getElementById('amount').value = '';
            document.getElementById('description').value = '';
        }

        function getEditableAccount(id) {
            if (id.startsWith('p_')) return players.find(p => p.id === id.substring(2));
            if (id.startsWith('acc_')) {
                const accId = id.substring(4);
                if (accId === accounts.mainCredit.id) return accounts.mainCredit;
                if (accId === accounts.mainBankReserve.id) return accounts.mainBankReserve; 
                return accounts.debit.find(acc => acc.id === accId);
            }
            return null;
        }


        function modifyBalance() {
            const accountId = document.getElementById('modify-account').value;
            const newBalanceInput = document.getElementById('new-balance').value;
            const newBalance = parseFloat(newBalanceInput);


            if (isNaN(newBalance)) {

                });


                alert(`${account.name} balance successfully set to $${newBalance.toLocaleString()}`);
                document.getElementById('new-balance').value = '';
                updateUI();
            } else {
                alert("Selected account not found.");
            }
        }

        function applyInterest() {
            const accountId = document.getElementById('apply-interest-account').value;
            const account = getEditableAccount(accountId);


            if (!account) {
                alert("Please select a valid account.");
                return;
            }


            const principal = account.balance;
            const rate = account.interestRate / 100;
            
            const interestAmount = toFixedMoney(principal * rate * 1);
            
            if (interestAmount === 0) {
                alert(`Interest amount is $0.00 (Balance: $${principal}, Rate: ${account.interestRate}%)`);
                return;
            }
            
            const reserve = accounts.mainBankReserve;


            if (interestAmount > 0 && reserve.balance < interestAmount) {
                alert(`Cannot pay interest! The ${reserve.name} only has $${reserve.balance.toLocaleString()} and needs $${interestAmount.toLocaleString()}.`);
                return;
            }


            account.balance += interestAmount;
            reserve.balance -= interestAmount;


            const type = interestAmount < 0 ? 'Interest Charge' : 'Interest Gain';
            const sender = interestAmount < 0 ? account.name : reserve.name;
            const receiver = interestAmount < 0 ? reserve.name : account.name;


            transactions.push({
                time: new Date().toLocaleTimeString(),
                sender: sender,
                receiver: receiver,
                amount: Math.abs(interestAmount),
                description: `${type} applied (${account.interestRate}% annual rate)`,
            });


            alert(`${type} of $${Math.abs(interestAmount).toLocaleString()} applied to ${account.name}. New Balance: $${account.balance.toLocaleString()}`);
            updateUI();
        }
